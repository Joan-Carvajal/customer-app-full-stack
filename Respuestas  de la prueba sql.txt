Segunda parte:
1. Listado de facturas registradas en todo el año 2017
SELECT *
FROM dbo.Facturas
WHERE YEAR(FechaRegistro) = 2017;

2. Listado de facturas de exportación con IVA mayor de 2.000.000
SELECT *
FROM dbo.Facturas
WHERE Exportacion = 1 
  AND ValorIVA > 2000000;

3. Listado de facturas del vendedor kperez que superaron los 10.000.000 de valor total
SELECT *
FROM dbo.Facturas
WHERE Vendedor = 'kperez' 
  AND ValorTotal > 10000000;

4. Listado de facturas registradas por el usuario rsolano
SELECT *
FROM dbo.Facturas
WHERE Usuario = 'rsolano';

5. Ventas totales por bodegas
SELECT 
    Bodega,
    SUM(ValorTotal) AS VentasTotales
FROM dbo.Facturas
GROUP BY Bodega
ORDER BY VentasTotales DESC;

6. Listado de productos de la factura con NumFactura = S00256, ordenado por cantidad de mayor a menor
SELECT 
    df.Producto,
    df.Cantidad,
    df.Valor,
    df.Lote
FROM dbo.DetalleFactura df
INNER JOIN dbo.Facturas f ON df.IdFactura = f.Id
WHERE f.NumFactura = 'S00256'
ORDER BY df.Cantidad DESC;

7. Cantidad de productos por facturas de mayo del 2018
SELECT 
    f.NumFactura,
    f.Cliente,
    COUNT(df.Id) AS CantidadProductos,
    SUM(df.Cantidad) AS CantidadTotal
FROM dbo.Facturas f
INNER JOIN dbo.DetalleFactura df ON f.Id = df.IdFactura
WHERE YEAR(f.FechaRegistro) = 2018 
  AND MONTH(f.FechaRegistro) = 5
GROUP BY f.NumFactura, f.Cliente
ORDER BY CantidadProductos DESC;

8. Listado de productos de las facturas del mes de enero del 2018
SELECT DISTINCT
    df.Producto,
    SUM(df.Cantidad) AS CantidadTotal,
    SUM(df.Valor * df.Cantidad) AS ValorTotal
FROM dbo.DetalleFactura df
INNER JOIN dbo.Facturas f ON df.IdFactura = f.Id
WHERE YEAR(f.FechaRegistro) = 2018 
  AND MONTH(f.FechaRegistro) = 1
GROUP BY df.Producto
ORDER BY CantidadTotal DESC;



10. Promedio de ventas por años
SELECT 
    YEAR(FechaRegistro) AS Anio,
    COUNT(*) AS NumeroFacturas,
    AVG(ValorTotal) AS PromedioVentas,
    SUM(ValorTotal) AS TotalVentas
FROM dbo.Facturas
GROUP BY YEAR(FechaRegistro)
ORDER BY Anio DESC;
